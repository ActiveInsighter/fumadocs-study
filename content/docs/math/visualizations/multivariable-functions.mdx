## 二重极限不存在的反例

<MathPlot3D
  title="f(x, y) = 2xy / (x² + y²)"
  description="沿直线 y = kx 趋于原点，函数值恒为 2k/(1 + k²)，与 k 有关，所以二重极限不存在。曲面在原点附近像一把被拧过的扇子。"
  x={[-1.2, 1.2]}
  y={[-1.2, 1.2]}
  z={[-1.1, 1.1]}
  aspect="cube"
  params={[{ name: 'k', label: 'k', min: -3, max: 3, value: 1 }]}
  items={[
    { type: 'surface', z: '2 x y / (x^2 + y^2)', r: [0.001, 1.2], resolution: 96, label: 'z=\\frac{2xy}{x^2+y^2}' },
    { type: 'curve', x: 't / sqrt(1 + k^2)', y: 'k t / sqrt(1 + k^2)', z: '2k / (1 + k^2)', t: [-1.2, 1.2], color: 'coral', tube: true, label: 'y=kx' },
  ]}
/>

<MathPlot3D
  title="f(x, y) = x²y / (x⁴ + y²)"

  ]}
